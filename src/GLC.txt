S -> DECL | BLOCO | NULO

NULO -> ; | S
BLOCO -> begin CMD* end | S

DECL -> (final | TYPE) id [,id]* (= const)?; | S

TYPE -> int | byte | string | boolean
ATTR -> id = (EXP | id);

EXPA -> '('? (id | const) OPMATH (id | const) ')'? (OPMATH EXPA)?;
EXPL -> '('? (id | const) OPLOG (id | const) ')'? (OPLOG EXPL)?;
OPLOG -> and | or | not | == | < | > | <> | >= | <=
OPMATH -> + | - | * | /


CMD -> 
	(
	id = (const); |
	while EXPL BLOCO |
	if EXPL BLOCO (else if EXPL BLOCO)* (else BLOCO)? |
	readln, id |
	write (EXPA [,EXPA]* | id [,id]*) |
	writeln (EXPA [,EXPA]* | id [,id]*)
	) | S;
