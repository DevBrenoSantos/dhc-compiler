S -> DECL* BLOCO

NULO -> ; 														// Tópico 7
BLOCO -> begin CMD* end									 		// Tópico 3

DECL -> (final | TYPE) id [= const] [,id [= const]]*;    		// Tópico 1,2 - (int a=5,b,c) 

TYPE -> int | byte | string | boolean
ATTR -> id = (id | const); 										// Tópico 4


EXPA -> SUM								// recursão baseado na precedência
	EXPA_SUM -> TERM [ (+ | -) TERM ]*
	EXPA_TERM -> UNARY [ ("*" | "/") UNARY ]*;
	EXPA_UNARY -> ["+" | "-"] NUMBER;
	NUMBER -> id | const | "(" EXPA ")";

EXPL -> EXPL_OR 						// recursão baseado na precedência
	EXPL_REL -> (EXPA | const | id) OPLOG (EXPA | const | id) | "(" EXPL ")"
	BOOL_VALUE -> EXPL_REL | id | const | "(" EXPL ")" | "(" EXPA ")"	
	EXPL_NOT -> [not] BOOL_VALUE
	EXPL_AND -> EXPL_NOT [and EXPL_NOT]*
	EXPL_OR -> EXPL_AND [or EXPL_AND]*		

OPLOG -> == | < | > | <> | >= | <=
OPMATH -> + | - | * | /
ANDOR -> and | or 

CMD -> (
	NULO
	ATTR |
	while EXPL BLOCO | 											// Tópico 5
	if EXPL BLOCO (else if EXPL BLOCO)* [else BLOCO] |			// Tópico 6
	readln, id |												// Tópico 8 
	write (id | const | EXPA) [,(id | const | EXPA)]* |			// Tópico 9
	writeln (id | const | EXPA) [id | const | EXPA]*\n
);


*se 'true' e 'false' são considerados const, EXPL sempre produz uma const
*espera-se que EXPL e EXPA sempre produzem uma const
