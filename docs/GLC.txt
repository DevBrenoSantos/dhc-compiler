S -> DECL | BLOCO | NULO

NULO -> ; 		// Tópico 7
BLOCO -> begin CMD* end  | begin  end 					 // Tópico 3

DECL -> (final | TYPE) id [= const] [,id [= const]]*;    // Tópico 1,2 - (int a=5,b,c) 

TYPE -> int | byte | string | boolean
ATTR -> id = (EXP | id | const); 			// Tópico 4

EXPA -> (TYPE)?  (id | const) (OPMATH (id | const))+ ;     // lembrar da possibilidade de parênteses no inicio e no fim da Expressão aritmética e lógica 
EXPL -> (id | const) OPLOG (id | const) (ANDOR EXPL)* ;			// Expressão Lógica
OPLOG -> not | == | < | > | <> | >= | <=
OPMATH -> + | - | * | /
ANDOR -> and | or 


CMD -> 
	(
	ATTR |
	while EXPL BLOCO | 					// Tópico 5
	if EXPL BLOCO (else if EXPL BLOCO)* (else BLOCO)? |			// Tópico 6
	readln, id |			// Tópico 8 
	write (id | const | EXPA) [,(id | const | EXPA)]* |				// Tópico 9
	writeln (id | const | EXPA) [id | const | EXPA]*\n
	);
